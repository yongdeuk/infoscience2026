    가운데 줄은 for _ in range(size - 2):로 반복하고, 공백 개수도 size - 2예요.

    정답 코드 보기
    def draw_square(size):
    print('#' * size)
    for _ in range(size - 2):
        print('#' + ' ' * (size - 2) + '#')
    print('#' * size)

draw_square(4)
print()
draw_square(5)
