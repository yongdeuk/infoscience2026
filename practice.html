def balanced(text):
    pair = {')': '(', ']': '[', '}': '{'}
    st = []
    for ch in text:
        if ch in '([{':
            st.append(ch)
        elif ch in pair:
            if not st or st.pop() != pair[ch]:
                return False
    return not st

for t in ['(a+b)*[c]', '((a)', '{[()]}', '(]']:

class Node:
    def __init__(self, v):
        self.v, self.left, self.right = v, None, None

def insert(root, v):
    if root is None:
        return Node(v)
    if v < root.v:
        root.left = insert(root.left, v)
    else:
        root.right = insert(root.right, v)
    return root

def inorder(node, out):
    if node:
        inorder(node.left, out)
        out.append(node.v)
        inorder(node.right, out)
    return out

root = None
for v in [50, 30, 70, 20, 40, 60, 80]:
    root = insert(root, v)
print('중위 순회:', inorder(root, []))

from collections import deque

def shortest(start, goal):
    prev = {start: None}
    q = deque([start])
    while q:
        node = q.popleft()
        if node == goal:
            break
        for nxt in graph[node]:
            if nxt not in prev:
                prev[nxt] = node
                q.append(nxt)
    path, cur = [], goal
    while cur is not None:
        path.append(cur)
        cur = prev[cur]
    return path[::-1]

print('A → H:', ' → '.join(shortest('A', 'H')))

2. 순차 탐색 vs 이진 탐색 — 비교 횟수를 비교합니다.

def seq_search(arr, t):
    for i, v in enumerate(arr):
        if v == t:
            return i + 1          # 비교 횟수
    return len(arr)

def bin_search(arr, t):
    lo, hi, c = 0, len(arr) - 1, 0
    while lo <= hi:
        mid = (lo + hi) // 2
        c += 1
        if arr[mid] == t:
            return c
        if arr[mid] < t:
            lo = mid + 1
        else:
            hi = mid - 1
    return c

data = list(range(1, 100001))
for t in [7, 50000, 99999]:
    print(f'{t:6}: 순차 {seq_search(data, t):6}번, 이진 {bin_search(data, t):2}번')

def greedy_change(amount, coins):
    used = []
    for c in sorted(coins, reverse=True):
        while amount >= c:
            amount -= c
            used.append(c)
    return used

def quick(arr):
    if len(arr) <= 1:
        return arr
    pivot = arr[0]
    small = [x for x in arr[1:] if x < pivot]
    big = [x for x in arr[1:] if x >= pivot]
    return quick(small) + [pivot] + quick(big)

print(quick([33, 10, 55, 71, 29, 3, 18, 42]))

2. 합병 정렬 — 반으로 나눠 정렬한 뒤, 두 정렬된 리스트를 합칩니다.

def merge_sort(arr):
    if len(arr) <= 1:
        return arr
    mid = len(arr) // 2
    a, b = merge_sort(arr[:mid]), merge_sort(arr[mid:])
    out, i, j = [], 0, 0
    while i < len(a) and j < len(b):
        if a[i] <= b[j]:
            out.append(a[i]); i += 1
        else:
            out.append(b[j]); j += 1
    return out + a[i:] + b[j:]

print(merge_sort([33, 10, 55, 71, 29, 3, 18, 42]))

1. 다익스트라 — 탄소 배출이 가장 적은 경로 — 가중치는 구간별 배출량(g)입니다.

import heapq

road = {
    '집':   {'정류장': 30, '역': 120},
    '정류장': {'역': 40, '학교': 150},
    '역':   {'학교': 35},
    '학교': {},
}

def dijkstra(start):
    dist = {n: float('inf') for n in road}
    dist[start] = 0
    pq = [(0, start)]
    while pq:
        d, node = heapq.heappop(pq)       # 가장 가까운 노드 확정
        if d > dist[node]:
            continue
        for nxt, w in road[node].items():
            if d + w < dist[nxt]:
                dist[nxt] = d + w
                heapq.heappush(pq, (dist[nxt], nxt))
    return dist

print(dijkstra('집'))

2. 백트래킹 — 짝꿍 배치 — 서로 사이가 안 좋은 두 학생이 옆자리에 앉지 않게 4자리에 배치합니다.

students = ['가온', '나래', '다솜', '라온']
avoid = {('가온', '나래'), ('다솜', '라온')}

def ok(seat, s):
    if not seat:
        return True
    left = seat[-1]
    return (left, s) not in avoid and (s, left) not in avoid

def place(seat, rest, out):
    if not rest:
        out.append(seat[:])
        return
    for s in rest:
        if ok(seat, s):                  # 조건을 어기면 이 가지는 버림
            seat.append(s)
            place(seat, [r for r in rest if r != s], out)
            seat.pop()                   # 되돌아가기
    return out

for arr in place([], students, []):
    print(arr)

IndentationError들여쓰기가 맞지 않아요. 블록 안의 줄은 같은 칸만큼 들여써야 해요.스페이스 4칸으로 통일하고, 탭과 스페이스를 섞지 않습니다.
NameError정의되지 않은 이름을 썼어요. 오타이거나, 그 이름을 만든 코드를 아직 실행하지 않았어요.철자·대소문자를 확인하고, 이 사이트에서는 위 칸부터 다시 실행합니다.
TypeError: can only concatenate str문자열과 숫자를 +로 붙이려 했어요.str(n)으로 바꾸거나 f-string f'{n}점'을 씁니다.
TypeError: missing 1 required positional argument함수에 필요한 인수를 덜 넣었어요. 메서드라면 self를 빠뜨린 경우도 있어요.함수 정의의 매개 변수 개수와 호출할 때의 인수 개수를 맞춥니다.
ValueError: invalid literal for int()숫자가 아닌 글자를 int()로 바꾸려 했어요.input()에 숫자만 넣었는지, 앞뒤 공백이 없는지 확인합니다.
IndexError: list index out of range리스트에 없는 위치를 꺼내려 했어요. 길이가 5면 인덱스는 0~4예요.range(len(a))를 쓰고, 빈 리스트에서 pop()·[0]을 하지 않았는지 봅니다.
IndexError: pop from empty list빈 스택·큐에서 꺼내려 했어요(언더플로).꺼내기 전에 if stack:으로 비었는지 먼저 확인합니다.
KeyError딕셔너리에 없는 키를 찾았어요.d.get(key, 기본값)을 쓰거나 if key in d:로 확인합니다.
ZeroDivisionError0으로 나누었어요. 빈 리스트의 평균을 구할 때 자주 나요.나누기 전에 분모가 0인지 확인합니다.
AttributeError그 객체에 없는 속성·메서드를 썼어요. 예: 리스트에 .push().철자를 확인하고, 리스트는 append, deque는 popleft처럼 맞는 이름을 씁니다.
RecursionError: maximum recursion depth재귀 호출이 너무 깊어졌어요. 대부분 종료 조건이 없거나 종료 조건에 닿지 않아요.종료 조건을 확인하고, 재귀 호출 때 n이 줄어드는지 봅니다.
UnboundLocalError함수 안에서 전역 변수 값을 바꾸려 했어요.함수 첫 줄에 global 변수이름을 쓰거나, 값을 매개 변수·반환값으로 주고받습니다.
ModuleNotFoundError설치되지 않은 라이브러리를 import했어요. Programiz에서는 numpy·matplotlib 같은 외부 라이브러리를 쓸 수 없어요.표준 라이브러리(random, math, collections, heapq 등)로 바꾸거나, 모듈 이름 오타를 확인합니다.
무한 반복(끝나지 않음)while 조건이 계속 참이에요.Programiz는 Stop(Run 자리) 버튼, 이 사이트는 새로 고침. 반복 안에서 조건 변수가 바뀌는지 확인합니다.
입력을 기다리며 멈춤(input)프로그램이 input()에서 값을 기다리는 중이에요.Programiz는 오른쪽 Output 창을 눌러 값을 넣고 Enter. 이 사이트에서는 실행 전에 입력값을 미리 적습니다.
NameError (Programiz)앞 칸에서 만든 함수·변수를 빼고 일부 코드만 붙여넣었어요.실습 칸 아래 Programiz에서 실행 버튼으로 다시 복사하면 앞 칸 코드까지 함께 들어갑니다.
예시 코드가 같이 실행됨Programiz 편집 창의 기본 예시 코드를 지우지 않고 붙여넣었어요.편집 창에서 Ctrl+A로 모두 선택한 뒤 붙여넣습니다.
