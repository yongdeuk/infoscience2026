    def get_result(user, comp):
    wins = {'가위': '보', '바위': '가위', '보': '바위'}   # 내가 이기는 상대
    if user == comp:
        return '비김'

        return '이김'

        return '짐'

score = 0
users = ['가위', '보', '바위', '보', '가위']
comps = ['보', '보', '보', '가위', '바위']

def play_game():
    global score
    for i in range(len(users)):
        result = get_result(users[i], comps[i])
        print(f'{i + 1}판: 나 {users[i]} vs 컴퓨터 {comps[i]} → {result}')
        if result == '이김':
            score += 1
    print(f'총점: {score}점')

play_game()
